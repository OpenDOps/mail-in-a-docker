1. Get the application URL by running these commands:
{{- if .Values.gateway.enabled }}
{{- $primaryHostname := .Values.mailinabox.env.primaryHostname }}
  Gateway is enabled. Access your Mail-in-a-Box at:
  - HTTP:  http://{{ $primaryHostname }}
  - HTTPS: https://{{ $primaryHostname }}

  Mail protocols:
  - SMTP:   {{ $primaryHostname }}:25
  - IMAP:   {{ $primaryHostname }}:143
  - IMAPS:  {{ $primaryHostname }}:993
  - SMTPS:  {{ $primaryHostname }}:465
  - Submission: {{ $primaryHostname }}:587

  {{- if .Values.gateway.tls }}
  {{- range $tls := .Values.gateway.tls }}
  {{- $hostname := $tls.hostname | default $primaryHostname }}
  Additional domain: {{ $hostname }} (TLS secret: {{ $tls.secretName }})
  {{- end }}
  {{- end }}
{{- else }}
  Gateway is disabled. Configure gateway.enabled=true to enable external access.
{{- end }}

2. Check certificate status:
{{- if .Values.certManager.enabled }}
  kubectl get certificate -n {{ .Release.Namespace }}
{{- end }}

3. Check Gateway status:
{{- if .Values.gateway.enabled }}
  kubectl get gateway -n {{ .Release.Namespace }}
  kubectl get httproute -n {{ .Release.Namespace }}
  kubectl get tcproute -n {{ .Release.Namespace }}
  kubectl get tlsroute -n {{ .Release.Namespace }}
{{- end }}
